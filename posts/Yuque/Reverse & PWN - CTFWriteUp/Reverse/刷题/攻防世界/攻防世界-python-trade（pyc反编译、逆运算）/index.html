import base64

def encode(message):

s = ‘’

for i in message:

x = ord(i) ^ 32

x = x + 16

s += chr(x)

return base64.b64encode(s)

correct = ‘XlNkVmtUI1MgXWBZXCFeKY+AaXNt’

flag = ‘’

print ‘Input flag:’

flag = raw_input()

if encode(flag) == correct:

print ‘correct’

else:

print ‘wrong’

ord()函数主要用来返回对应字符的ascii码，chr()主要用来表示ascii码对应的字符他的输入时数字，可以用十进制，也可以用十六进制。

例如：print ord(‘a)

         #97

         print chr(97)

        #a

        print chr(0x61)

       #a

也可以用如下代码直接进行解密

pip install uncompyle

uncompyle6 11.pyc > 11.py

这个程序大概的意思是让你输入字符串经过一系列的运算和加密之后得到XlNkVmtUI1MgXWBZXCFeKY+AaXNt

那就进行逆运算吧，但是不会写python脚本。。。

在网上copy一个

import base64//导入base64库

s=’XlNkVmtUI1MgXWBZXCFeKY+AaXNt’//字符串用单引号括起来

str=base64.b64decode(s)//解密XlNkVmtUI1MgXWBZXCFeKY+AaXNt

flag=’’//flag为NULL

for i in str:

i-=16//逆运算

i^=32//对某个值进行两次异或运算得到的还是它本身

flag+=chr(i)//逆运算

print(flag)
